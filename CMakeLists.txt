target_include_directories(tetorica_spu_instrument SYSTEM PRIVATE
  vendor/iPlug2OOS/iPlug2/Dependencies/Extras)
target_link_libraries(tetorica_spu_instrument PUBLIC tetorica::ps1_spu tetorica_vab Threads::Threads)
# Keep cyclic encoder scoring and Morph blending equivalent to the Web's
# separate IEEE754 operations on supported toolchains.
if(MSVC)
  set_source_files_properties(src/SpuWaveform.cpp src/SpuMorph.cpp src/WebInstrument.cpp src/WavImport.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  set_source_files_properties(src/SpuWaveform.cpp src/SpuMorph.cpp src/WebInstrument.cpp src/WavImport.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
endif()

option(TETORICA_BUILD_PLUGIN "Build iPlug2 plugin bundles" ${_plugin_default})
if(TETORICA_BUILD_PLUGIN)
  if(NOT CMAKE_OSX_DEPLOYMENT_TARGET AND APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
  endif()
  # Never auto-install or overwrite plugins in the user's home directory.
  set(IPLUG_DEPLOY_PLUGINS OFF CACHE BOOL "No automatic plugin deployment" FORCE)
  set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/iPlug2OOS/iPlug2")
  if(NOT EXISTS "${IPLUG2_DIR}/Dependencies/IPlug/VST3_SDK/pluginterfaces/base/funknown.h")
    message(FATAL_ERROR "Missing VST3 SDK: run python3 scripts/bootstrap.py")
  endif()
  include(${IPLUG2_DIR}/iPlug2.cmake)
  find_package(iPlug2 REQUIRED)
  add_subdirectory(plugin)
endif()
