cmake_minimum_required(VERSION 3.25)
project(tetorica_spu_plugin VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(TETORICA_NATIVE_BUILD_TESTS OFF CACHE BOOL "SDK tests built independently" FORCE)
set(TETORICA_NATIVE_BUILD_EXAMPLES OFF CACHE BOOL "SDK examples built independently" FORCE)
# Keep SDK-internal CTest fixtures out of this consumer's test graph. Some
# targets use BUILD_TESTING rather than TETORICA_NATIVE_BUILD_TESTS.
set(_consumer_build_testing ${BUILD_TESTING})
set(BUILD_TESTING OFF)
add_subdirectory(vendor/tetorica-soundchip-native EXCLUDE_FROM_ALL)
set(BUILD_TESTING ${_consumer_build_testing})

add_library(tetorica_vab STATIC src/VabBank.cpp)
target_include_directories(tetorica_vab PUBLIC src)

add_library(tetorica_spu_instrument STATIC src/SpuInstrument.cpp)
target_include_directories(tetorica_spu_instrument PUBLIC src)
target_link_libraries(tetorica_spu_instrument PUBLIC tetorica::ps1_spu tetorica_vab)

add_executable(spu_vab_render tools/spu-vab-render.cpp)
target_link_libraries(spu_vab_render PRIVATE tetorica_spu_instrument)

if(BUILD_TESTING)
  add_executable(spu_instrument_test tests/spu_instrument_test.cpp)
  target_link_libraries(spu_instrument_test PRIVATE tetorica_spu_instrument)
  add_test(NAME spu.instrument COMMAND spu_instrument_test)
  add_executable(vab_bank_test tests/vab_bank_test.cpp)
  target_link_libraries(vab_bank_test PRIVATE tetorica_vab)
  target_compile_definitions(vab_bank_test PRIVATE VAB_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/vab")
  add_test(NAME spu.vab.bank COMMAND vab_bank_test)
  add_executable(spu_vab_instrument_test tests/spu_vab_instrument_test.cpp)
  target_link_libraries(spu_vab_instrument_test PRIVATE tetorica_spu_instrument)
  target_compile_definitions(spu_vab_instrument_test PRIVATE VAB_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/vab")
  add_test(NAME spu.vab.instrument COMMAND spu_vab_instrument_test)
endif()

if(APPLE OR WIN32)
  set(_plugin_default ON)
else()
  set(_plugin_default OFF)
endif()
option(TETORICA_BUILD_PLUGIN "Build iPlug2 plugin bundles" ${_plugin_default})
if(TETORICA_BUILD_PLUGIN)
  if(NOT CMAKE_OSX_DEPLOYMENT_TARGET AND APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
  endif()
  # Never auto-install or overwrite plugins in the user's home directory.
  set(IPLUG_DEPLOY_PLUGINS OFF CACHE BOOL "No automatic plugin deployment" FORCE)
  set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/iPlug2OOS/iPlug2")
  if(NOT EXISTS "${IPLUG2_DIR}/Dependencies/IPlug/VST3_SDK/pluginterfaces/base/funknown.h")
    message(FATAL_ERROR "Missing VST3 SDK: run python3 scripts/bootstrap.py")
  endif()
  include(${IPLUG2_DIR}/iPlug2.cmake)
  find_package(iPlug2 REQUIRED)
  add_subdirectory(plugin)
endif()
